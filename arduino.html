--- ## ☁️ Firebase Setup 1. Go to **[Firebase Console](https://console.firebase.google.com/)**. 2. Click **Add Project → Create New Project**. 3. Navigate to **Realtime Database → Create Database**. 4. Choose your region and **start in test mode**. 5. Copy: - **Database URL** → e.g. `https://smart-classroom-esp32-default-rtdb.firebaseio.com/` - **Database Secret** → Go to *Project Settings → Service Accounts → Database Secrets*. --- ## 💻 Arduino Code ```cpp #include #include // ---------- WIFI CONFIG ------------ #define WIFI_SSID "Your_WiFi_Name" #define WIFI_PASSWORD "Your_WiFi_Password" // ---------- FIREBASE CONFIG ---------- #define FIREBASE_HOST "your-project.firebaseio.com" #define FIREBASE_AUTH "your-firebase-secret" FirebaseData firebaseData; // ---------- PIN CONFIG --------------- #define PIR_PIN 13 #define LDR_PIN 34 #define RELAY_PIN 26 void setup() { Serial.begin(115200); pinMode(PIR_PIN, INPUT); pinMode(LDR_PIN, INPUT); pinMode(RELAY_PIN, OUTPUT); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); Serial.print("Connecting to WiFi"); while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); } Serial.println("\n✅ Connected to WiFi"); Firebase.begin(FIREBASE_HOST, FIREBASE_AUTH); Firebase.reconnectWiFi(true); Serial.println("✅ Connected to Firebase"); } void loop() { int motion = digitalRead(PIR_PIN); int lightLevel = analogRead(LDR_PIN); Serial.print("Motion: "); Serial.print(motion); Serial.print(" | Light: "); Serial.println(lightLevel); // Control logic: Light ON if motion detected and room is dark if (motion == HIGH && lightLevel < 1000) { digitalWrite(RELAY_PIN, HIGH); Firebase.setString(firebaseData, "/classroom/status", "ON"); Firebase.setInt(firebaseData, "/classroom/lightLevel", lightLevel); Firebase.setString(firebaseData, "/classroom/motion", "Detected"); } else { digitalWrite(RELAY_PIN, LOW); Firebase.setString(firebaseData, "/classroom/status", "OFF"); Firebase.setInt(firebaseData, "/classroom/lightLevel", lightLevel); Firebase.setString(firebaseData, "/classroom/motion", "No Motion"); } delay(2000); }
